const SHEET_NAMES = {
  habits:     'Habits',
  satish:     'Satish',
  keerthisri: 'KeerthiSri',
  geetanath:  'Geetanath'
};

// ── CORS wrapper ────────────────────────────────────────────────
function doGet(e)  { return handle(e); }
function doPost(e) { return handle(e); }

function handle(e) {
  try {
    const params = e.parameter || {};
    const body   = e.postData ? JSON.parse(e.postData.contents) : {};
    const action = params.action || body.action;
    let result;

    switch (action) {
      case 'getAll':       result = getAll();                    break;
      case 'saveHabit':    result = saveHabit(body.habit);       break;
      case 'deleteHabit':  result = deleteHabit(body.habitId);   break;
      case 'toggleCompletion': result = toggleCompletion(body);  break;
      case 'saveQtyLog':   result = saveQtyLog(body);            break;
      case 'savePeriodData': result = savePeriodData(body.periodDates); break;
      case 'saveMoodData':   result = saveMoodData(body.moods);  break;
      default:             result = { error: 'Unknown action: ' + action };
    }

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, data: result }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ── Sheet bootstrap ─────────────────────────────────────────────
function getOrCreateSheet(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(name);
  
  if (!sheet) {
    sheet = ss.insertSheet(name);
    // Set headers based on sheet type
    if (name === SHEET_NAMES.habits) {
      sheet.appendRow(['id','name','emoji','freq','color','assignee','assignees','cpN','cpPeriod','type','qtyUnit','qtyGoal','createdAt']);
      sheet.getRange(1,1,1,13).setFontWeight('bold').setBackground('#e8f5e3');
    } else {
      // Assignee completion sheets: habitId | date | done | qtyValue
      sheet.appendRow(['habitId','date','done','qtyValue']);
      sheet.getRange(1,1,1,4).setFontWeight('bold').setBackground('#e8f5e3');
    }
    sheet.setFrozenRows(1);
  } else if (name === SHEET_NAMES.habits) {
    // Existing Habits sheet - check and add missing columns
    ensureHabitsColumns(sheet);
  }
  
  return sheet;
}

// Ensure Habits sheet has all required columns
function ensureHabitsColumns(sheet) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const requiredHeaders = ['id','name','emoji','freq','color','assignee','assignees','cpN','cpPeriod','type','qtyUnit','qtyGoal','createdAt'];
  const missingHeaders = requiredHeaders.filter(h => !headers.includes(h));
  
  if (missingHeaders.length > 0) {
    Logger.log('Adding missing columns to Habits sheet: ' + missingHeaders.join(', '));
    const startCol = headers.length + 1;
    missingHeaders.forEach((header, i) => {
      sheet.getRange(1, startCol + i).setValue(header);
    });
    // Format header row
    sheet.getRange(1, 1, 1, headers.length + missingHeaders.length)
      .setFontWeight('bold')
      .setBackground('#e8f5e3');
  }
}

// ── GET ALL DATA ─────────────────────────────────────────────────
// Returns { habits: [...], completions: { habitId: { dateKey: true } }, qtyLogs: { habitId: { dateKey: number } } }
function getAll() {
  const habitsSheet = getOrCreateSheet(SHEET_NAMES.habits);
  const habitsData  = habitsSheet.getDataRange().getValues();
  const headers     = habitsData[0];

  const habits = habitsData.slice(1).filter(r => r[0]).map(row => {
    const obj = {};
    headers.forEach((h, i) => {
      // Convert empty strings to null for consistency
      obj[h] = row[i] === '' ? null : row[i];
    });
    
    // Log raw data for first habit to debug
    if (habitsData.indexOf(row) === 1) {
      Logger.log('First habit raw data: ' + JSON.stringify(obj));
    }
    
    // Parse numeric fields
    if (obj.cpN !== null && obj.cpN !== undefined) obj.cpN = Number(obj.cpN);
    if (obj.qtyGoal !== null && obj.qtyGoal !== undefined && obj.qtyGoal !== '') obj.qtyGoal = Number(obj.qtyGoal);
    
    // Parse assignees array (stored as JSON string)
    if (obj.assignees && typeof obj.assignees === 'string') {
      try {
        obj.assignees = JSON.parse(obj.assignees);
      } catch(e) {
        // Fallback: comma-separated string
        obj.assignees = obj.assignees.split(',').map(s => s.trim()).filter(Boolean);
      }
    }
    if (!obj.assignees || !Array.isArray(obj.assignees)) {
      // Fallback to single assignee
      obj.assignees = obj.assignee ? [obj.assignee] : [];
    }
    
    // Ensure type field exists (should be 'check' or 'qty')
    if (!obj.type) {
      obj.type = 'check';  // default
    }
    
    return obj;
  });
  
  Logger.log('Returning ' + habits.length + ' habits from getAll()');

  // Gather completions and quantity logs from all assignee sheets
  const completions = {};
  const qtyLogs = {};
  let qtyLogCount = 0;
  let skippedRowCount = 0;
  
  Object.values(SHEET_NAMES).forEach(sheetName => {
    if (sheetName === SHEET_NAMES.habits) return;
    const sheet = getOrCreateSheet(sheetName);
    const data  = sheet.getDataRange().getValues();
    
    // Ensure qtyValue column exists (migration for existing sheets)
    if (data[0].length === 3 && data[0].indexOf('qtyValue') === -1) {
      sheet.getRange(1, 4).setValue('qtyValue');
      sheet.getRange(1, 1, 1, 4).setFontWeight('bold').setBackground('#e8f5e3');
      Logger.log('Added qtyValue column to sheet: ' + sheetName);
    }
    
    data.slice(1).forEach((row, index) => {
      const [habitId, date, done, qtyValue] = row;
      
      // Skip rows with missing required fields
      if (!habitId || !date) {
        skippedRowCount++;
        if (!habitId && qtyValue) {
          Logger.log(`⚠️ Skipping row ${index + 2} in ${sheetName}: Missing habitId but has qtyValue=${qtyValue}`);
        }
        return;
      }
      
      const dateStr = formatDate(date);
      
      // Store completions (for check-type habits)
      if (done) {
        if (!completions[habitId]) completions[habitId] = {};
        completions[habitId][dateStr] = true;
      }
      
      // Store quantity logs (for qty-type habits)
      if (qtyValue !== undefined && qtyValue !== null && qtyValue !== '') {
        if (!qtyLogs[habitId]) qtyLogs[habitId] = {};
        qtyLogs[habitId][dateStr] = Number(qtyValue);
        qtyLogCount++;
      }
    });
  });

  Logger.log('Loaded ' + qtyLogCount + ' quantity logs' + (skippedRowCount > 0 ? ` (skipped ${skippedRowCount} invalid rows)` : ''));
  
  // Debug: Show what we're returning
  Logger.log('Returning data:');
  Logger.log('  - Habits: ' + habits.length);
  Logger.log('  - Completion habit IDs: ' + Object.keys(completions).length);
  Logger.log('  - QtyLog habit IDs: ' + Object.keys(qtyLogs).length);
  
  if (Object.keys(qtyLogs).length > 0) {
    Logger.log('QtyLogs details:');
    Object.keys(qtyLogs).forEach(habitId => {
      const habit = habits.find(h => h.id === habitId);
      const habitName = habit ? habit.name : 'Unknown';
      Logger.log('  - ' + habitName + ' (' + habitId + '): ' + Object.keys(qtyLogs[habitId]).length + ' entries');
      Object.keys(qtyLogs[habitId]).forEach(dateKey => {
        Logger.log('    - ' + dateKey + ': ' + qtyLogs[habitId][dateKey]);
      });
    });
  } else {
    Logger.log('  ⚠️ WARNING: qtyLogs is empty!');
  }
  
  // Verify the return object structure
  const returnObj = { habits: habits, completions: completions, qtyLogs: qtyLogs };
  Logger.log('Return object keys: ' + Object.keys(returnObj).join(', '));
  Logger.log('Return object.qtyLogs type: ' + typeof returnObj.qtyLogs);
  Logger.log('Return object.qtyLogs: ' + JSON.stringify(returnObj.qtyLogs));

  // Add period and mood tracker data from sheets
  // Look for period and mood data in the assignee sheets (stored with special habit IDs)
  const periodDates = {};
  const moods = { satish: {}, keerthisri: {}, geetanath: {} };
  
  Object.values(SHEET_NAMES).forEach(sheetName => {
    if (sheetName === SHEET_NAMES.habits) return;
    const sheet = getOrCreateSheet(sheetName);
    const data  = sheet.getDataRange().getValues();
    
    data.slice(1).forEach(row => {
      const [habitId, date, done, qtyValue] = row;
      if (!habitId || !date) return;
      const dateStr = formatDate(date);
      
      // Period tracker uses special habit ID 'period_tracker'
      if (habitId === 'period_tracker' && done) {
        periodDates[dateStr] = true;
      }
      
      // Mood tracker uses special habit ID pattern 'mood_<userId>'
      if (habitId && habitId.startsWith('mood_')) {
        const userId = habitId.replace('mood_', '');
        if (moods[userId] && qtyValue) {
          moods[userId][dateStr] = String(qtyValue);
        }
      }
    });
  });
  
  returnObj.periodDates = periodDates;
  returnObj.moods = moods;
  
  Logger.log('Return object now includes periodDates and moods');
  Logger.log('  - Period dates count: ' + Object.keys(periodDates).length);
  Logger.log('  - Mood entries: satish=' + Object.keys(moods.satish).length + 
             ', keerthisri=' + Object.keys(moods.keerthisri).length + 
             ', geetanath=' + Object.keys(moods.geetanath).length);

  return returnObj;
}

// ── SAVE HABIT ───────────────────────────────────────────────────
// Upserts a habit row in the Habits sheet
function saveHabit(habit) {
  const sheet   = getOrCreateSheet(SHEET_NAMES.habits);
  const data    = sheet.getDataRange().getValues();
  const headers = data[0];
  
  // Log the incoming habit for debugging
  Logger.log('Saving habit: ' + JSON.stringify(habit));
  
  // Remove local-only fields that shouldn't be saved to sheets
  const cleanHabit = Object.assign({}, habit);
  delete cleanHabit.synced;

  // Find existing row by id
  let rowIdx = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(habit.id)) { rowIdx = i + 1; break; }
  }

  const row = headers.map(h => {
    const v = cleanHabit[h];
    // Special handling for assignees array - store as JSON
    if (h === 'assignees' && Array.isArray(v)) {
      return JSON.stringify(v);
    }
    // Convert null/undefined to empty string for sheets
    if (v === undefined || v === null) return '';
    // For boolean, number, string - return as is
    return v;
  });
  
  // Create a debug object to see what we're saving
  const debugObj = {};
  headers.forEach((h, i) => {
    debugObj[h] = row[i];
  });
  Logger.log('Saving row with values: ' + JSON.stringify(debugObj));

  if (rowIdx > 0) {
    sheet.getRange(rowIdx, 1, 1, row.length).setValues([row]);
    Logger.log('Updated existing row ' + rowIdx);
  } else {
    sheet.appendRow(row);
    Logger.log('Appended new row');
  }
  return { saved: habit.id };
}

// ── DELETE HABIT ─────────────────────────────────────────────────
function deleteHabit(habitId) {
  // Remove from Habits sheet
  const hSheet = getOrCreateSheet(SHEET_NAMES.habits);
  const hData  = hSheet.getDataRange().getValues();
  for (let i = hData.length - 1; i >= 1; i--) {
    if (String(hData[i][0]) === String(habitId)) {
      hSheet.deleteRow(i + 1);
      break;
    }
  }

  // Remove completions from all assignee sheets
  Object.values(SHEET_NAMES).forEach(sheetName => {
    if (sheetName === SHEET_NAMES.habits) return;
    const sheet = getOrCreateSheet(sheetName);
    const data  = sheet.getDataRange().getValues();
    // Delete in reverse to preserve row indices
    for (let i = data.length - 1; i >= 1; i--) {
      if (String(data[i][0]) === String(habitId)) sheet.deleteRow(i + 1);
    }
  });

  return { deleted: habitId };
}

// ── TOGGLE COMPLETION ────────────────────────────────────────────
// body: { habitId, dateKey, assignee, value: true/false }
function toggleCompletion(body) {
  const { habitId, dateKey, assignee, value } = body;
  const sheetName = SHEET_NAMES[assignee] || SHEET_NAMES.satish;
  const sheet     = getOrCreateSheet(sheetName);
  const data      = sheet.getDataRange().getValues();

  // Find existing row
  let foundRow = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(habitId) && formatDate(data[i][1]) === dateKey) {
      foundRow = i + 1; break;
    }
  }

  if (value === true || value === 'true') {
    if (foundRow > 0) {
      sheet.getRange(foundRow, 3).setValue(true);
    } else {
      sheet.appendRow([habitId, dateKey, true, '']);
    }
  } else {
    // Mark as false or delete the row
    if (foundRow > 0) sheet.deleteRow(foundRow);
  }

  return { habitId, dateKey, value };
}

// ── SAVE QUANTITY LOG ────────────────────────────────────────────
// body: { habitId, dateKey, assignee, qtyValue: number }
function saveQtyLog(body) {
  const { habitId, dateKey, assignee, qtyValue } = body;
  
  // Validate required fields
  if (!habitId) {
    Logger.log('ERROR: saveQtyLog called without habitId');
    throw new Error('habitId is required for saveQtyLog');
  }
  if (!dateKey) {
    Logger.log('ERROR: saveQtyLog called without dateKey');
    throw new Error('dateKey is required for saveQtyLog');
  }
  
  const sheetName = SHEET_NAMES[assignee] || SHEET_NAMES.satish;
  const sheet     = getOrCreateSheet(sheetName);
  const data      = sheet.getDataRange().getValues();
  
  Logger.log('Saving qty log: habitId=' + habitId + ', dateKey=' + dateKey + ', assignee=' + assignee + ', qtyValue=' + qtyValue);

  // Find existing row
  let foundRow = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(habitId) && formatDate(data[i][1]) === dateKey) {
      foundRow = i + 1; break;
    }
  }

  if (qtyValue !== undefined && qtyValue !== null && qtyValue !== '') {
    if (foundRow > 0) {
      // Update existing row - set qtyValue in column 4t
      sheet.getRange(foundRow, 4).setValue(Number(qtyValue));
      Logger.log('Updated existing row ' + foundRow + ' with qtyValue: ' + qtyValue);
    } else {
      // Add new row: habitId, dateKey, done (empty for qty habits), qtyValue
      sheet.appendRow([habitId, dateKey, '', Number(qtyValue)]);
      Logger.log('Appended new row with qtyValue: ' + qtyValue);
    }
  } else if (foundRow > 0) {
    // Delete if qtyValue is empty/null
    sheet.deleteRow(foundRow);
    Logger.log('Deleted row ' + foundRow + ' (qtyValue was empty)');
  }

  return { habitId, dateKey, qtyValue };
}

// ── SAVE PERIOD TRACKER DATA ─────────────────────────────────────
// periodDates: { 'YYYY-MM-DD': true, ... }
function savePeriodData(periodDates) {
  if (!periodDates || typeof periodDates !== 'object') {
    Logger.log('ERROR: Invalid periodDates');
    return { error: 'Invalid period data' };
  }
  
  Logger.log('Saving period data: ' + Object.keys(periodDates).length + ' dates');
  
  // Use satish's sheet for period tracker (can use any assignee sheet)
  const sheet = getOrCreateSheet(SHEET_NAMES.satish);
  const data  = sheet.getDataRange().getValues();
  
  // Remove all existing period tracker entries
  for (let i = data.length - 1; i >= 1; i--) {
    if (String(data[i][0]) === 'period_tracker') {
      sheet.deleteRow(i + 1);
    }
  }
  
  // Add all period dates
  Object.keys(periodDates).forEach(dateKey => {
    if (periodDates[dateKey]) {
      sheet.appendRow(['period_tracker', dateKey, true, '']);
    }
  });
  
  Logger.log('Period data saved: ' + Object.keys(periodDates).length + ' dates');
  return { saved: Object.keys(periodDates).length };
}

// ── SAVE MOOD TRACKER DATA ───────────────────────────────────────
// moods: { userId: { 'YYYY-MM-DD': moodId, ... }, ... }
function saveMoodData(moods) {
  if (!moods || typeof moods !== 'object') {
    Logger.log('ERROR: Invalid moods');
    return { error: 'Invalid mood data' };
  }
  
  let totalSaved = 0;
  
  Object.keys(moods).forEach(userId => {
    if (!SHEET_NAMES[userId]) {
      Logger.log('WARNING: Unknown user ID: ' + userId);
      return;
    }
    
    const userMoods = moods[userId] || {};
    const sheet = getOrCreateSheet(SHEET_NAMES[userId]);
    const data  = sheet.getDataRange().getValues();
    
    Logger.log('Saving mood data for ' + userId + ': ' + Object.keys(userMoods).length + ' entries');
    
    // Remove all existing mood entries for this user
    for (let i = data.length - 1; i >= 1; i--) {
      if (String(data[i][0]) === 'mood_' + userId) {
        sheet.deleteRow(i + 1);
      }
    }
    
    // Add all mood entries for this user
    Object.keys(userMoods).forEach(dateKey => {
      const moodId = userMoods[dateKey];
      if (moodId) {
        sheet.appendRow(['mood_' + userId, dateKey, '', String(moodId)]);
        totalSaved++;
      }
    });
  });
  
  Logger.log('Mood data saved: ' + totalSaved + ' total entries');
  return { saved: totalSaved };
}

// ── HELPERS ──────────────────────────────────────────────────────
function formatDate(val) {
  if (!val) return '';
  if (typeof val === 'string') return val.substring(0, 10);
  // Date object from Sheets - use local date parts to avoid timezone issues
  const d = new Date(val);
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

// ── MIGRATION HELPER ─────────────────────────────────────────────
// Run this once manually if you have an existing Habits sheet
// It will add the missing columns: assignees, type, qtyUnit, qtyGoal
function migrateHabitsSheet() {
  const sheet = getOrCreateSheet(SHEET_NAMES.habits);
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  
  // Check which columns are missing
  const requiredHeaders = ['id','name','emoji','freq','color','assignee','assignees','cpN','cpPeriod','type','qtyUnit','qtyGoal','createdAt'];
  const missingHeaders = requiredHeaders.filter(h => !headers.includes(h));
  
  if (missingHeaders.length === 0) {
    Logger.log('No migration needed - all columns present');
    return;
  }
  
  Logger.log('Adding missing columns: ' + missingHeaders.join(', '));
  
  // Add missing columns to the end
  const startCol = headers.length + 1;
  missingHeaders.forEach((header, i) => {
    sheet.getRange(1, startCol + i).setValue(header);
  });
  
  // Format header row
  sheet.getRange(1, 1, 1, headers.length + missingHeaders.length)
    .setFontWeight('bold')
    .setBackground('#e8f5e3');
  
  Logger.log('Migration complete! Added ' + missingHeaders.length + ' columns.');
}

  let totalLogs = 0;
  Object.values(SHEET_NAMES).forEach(sheetName => {
    if (sheetName === SHEET_NAMES.habits) return;

      const status = issues.length > 0 ? issues.join(', ') : '✓ Valid';
      
      Logger.log(`Row ${i}: habitId="${habitId}", date="${formatDate(date)}", done="${done}", qtyValue="${qtyValue}" - ${status}`);
      
      if (habitId && date && (qtyValue !== undefined && qtyValue !== null && qtyValue !== '')) {
        totalLogs++;
      }
    }
  });
  
  Logger.log('\n=== SUMMARY ===');
  Logger.log('Total valid quantity logs: ' + totalLogs);
  Logger.log('\nIf you see rows with missing habitId, either:');
  Logger.log('1. Delete those rows, OR');
  Logger.log('2. Fill in the habitId using showAllHabitIds() to get valid IDs');
}
